{# Week navigator for the journal, like the calendar's: an arrow to the week before, the open week, an arrow to the week after, and a way back to this week. Folders are the main way to get around. #} {% macro journal_strip(weeks, active_week='') %} {% set _ns = namespace(i=-1, now=-1) %} {% for w in weeks %}{% if w.week == active_week %}{% set _ns.i = loop.index0 %}{% endif %}{% if w.is_current %}{% set _ns.now = loop.index0 %}{% endif %}{% endfor %} {% set _prev = weeks[_ns.i - 1] if _ns.i > 0 else none %} {% set _next = weeks[_ns.i + 1] if _ns.i >= 0 and _ns.i < weeks|length - 1 else none %} {% set _now = weeks[_ns.now] if _ns.now >= 0 and _ns.now != _ns.i else none %} {% macro step(w, side) %} {% set _name = 'Week ' ~ w.nr %} {% if w.uid %}{{ icon('chevron-left' if side == 'prev' else 'chevron-right', 'icon-sm') }} {% else %}{% endif %} {% endmacro %} {% if _prev or _next or _now %}
{% if _ns.i >= 0 %} {% if _prev %}{{ step(_prev, 'prev') }}{% else %}{{ icon('chevron-left', 'icon-sm') }}{% endif %} Week {{ weeks[_ns.i].nr }} {% if _next %}{{ step(_next, 'next') }}{% else %}{{ icon('chevron-right', 'icon-sm') }}{% endif %} {% endif %} {% if _now %} {% if _now.uid %}This week {% else %}{% endif %} {% endif %}
{% endif %} {% endmacro %}
